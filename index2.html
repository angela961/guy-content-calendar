// ── API calls via JSONP (works cross-origin from GitHub Pages) ─

function jsonpLoad(){

  return new Promise((resolve,reject)=>{

    const cb='cb_'+Date.now();

    const s=document.createElement('script');

    const timer=setTimeout(()=>{cleanup();reject(new Error('timeout'));},10000);

    window[cb]=(data)=>{cleanup();resolve(data);};

    function cleanup(){clearTimeout(timer);delete window[cb];s.parentNode&&s.parentNode.removeChild(s);}

    s.onerror=()=>{cleanup();reject(new Error('load error'));};

    s.src=SCRIPT_URL+'?action=load&callback='+cb;

    document.head.appendChild(s);

  });

}


async function postData(payload){

  try{

    await fetch(SCRIPT_URL,{

      method:'POST',

      mode:'no-cors',

      body:JSON.stringify(payload)

    });

  }catch(e){}

}


async function loadData(){

  setStatus('Cargando...','loading');

  try{

    const data=await jsonpLoad();

    if(data.ok){

      assignments=data.entries||{};

      setStatus('Sincronizado · '+new Date().toLocaleTimeString(),'ok');

    } else {

      throw new Error(data.error||'error');

    }

  }catch(e){

    setStatus('Error — revisa la URL del script','error');

    console.error(e);

  }

  renderCalendar();

}


// ── Calendar ─────────────────────────────────────────────────

function renderCalendar(){

  const grid=document.getElementById('cal-grid');

  grid.innerHTML='';

  DAY_NAMES.forEach((d,i)=>{

    const h=document.createElement('div');

    h.className='day-hdr'+(POSTING_DAYS.includes(i)?' posting':'');

    h.textContent=d;grid.appendChild(h);

  });

  const firstDay=new Date(viewYear,viewMonth,1).getDay();

  const daysInMonth=new Date(viewYear,viewMonth+1,0).getDate();

  for(let i=0;i<firstDay;i++){

    const b=document.createElement('div');b.className='day-cell off-day';grid.appendChild(b);

  }

  for(let d=1;d<=daysInMonth;d++){

    const jsDay=new Date(viewYear,viewMonth,d).getDay();

    const isPosting=POSTING_DAYS.includes(jsDay);

    const key=viewYear+'-'+String(viewMonth+1).padStart(2,'0')+'-'+String(d).padStart(2,'0');

    const assigned=assignments[key];

    const isToday=(today.getFullYear()===viewYear&&today.getMonth()===viewMonth&&today.getDate()===d);

    if(!isPosting){

      const cell=document.createElement('div');cell.className='day-cell off-day';

      const n=document.createElement('div');n.className='day-num';n.textContent=d;

      cell.appendChild(n);grid.appendChild(cell);continue;

    }

    const cell=document.createElement('div');

    cell.className='day-cell posting-day'+(assigned?' has-content':'');

    if(assigned){const p=getPillar(assigned.pillarId);if(p)cell.style.background=p.bg+'CC';}

    cell.onclick=()=>openModal(key,d);

    const numEl=document.createElement('div');

    numEl.className='day-num'+(isToday?' today-circle':' active-num');

    numEl.textContent=d;cell.appendChild(numEl);

    if(assigned){

      const p=getPillar(assigned.pillarId);

      if(p){

        const slot=document.createElement('div');slot.className='slot';slot.style.color=p.fg;

        const pl=document.createElement('div');pl.className='slot-pillar';pl.textContent=p.label;slot.appendChild(pl);

        if(assigned.copy){const cp=document.createElement('div');cp.className='slot-copy';cp.textContent=assigned.copy;slot.appendChild(cp);}

        const chs=document.createElement('div');chs.className='slot-channels';

        CHANNELS.forEach(ch=>{

          if(assigned.channels&&assigned.channels[ch.id]&&assigned.channels[ch.id].enabled){

            const b=document.createElement('div');b.className='slot-ch';b.style.color=p.fg;

            b.textContent=(ch.id==='fbig'?'FB/IG':ch.name)+' '+assigned.channels[ch.id].time;

            chs.appendChild(b);

        });

        slot.appendChild(chs);cell.appendChild(slot);

      }

    } else {

      const ab=document.createElement('div');ab.className='add-btn';ab.textContent='+ Schedule';cell.appendChild(ab);

    }

    grid.appendChild(cell);

  document.getElementById('month-label').textContent=MONTH_NAMES[viewMonth]+' '+viewYear;

}


// ── Modal ─────────────────────────────────────────────────────

function openModal(key,day){

  modalDate=key;

  const ex=assignments[key];

  draft=ex

    ?{pillarId:ex.pillarId,copy:ex.copy||'',channels:JSON.parse(JSON.stringify(ex.channels||freshChannels()))}

    :{pillarId:null,copy:'',channels:freshChannels()};

  renderModal(day);

}

function escHtml(s){return(s||'').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');}

function renderModal(day){

  const ex=assignments[modalDate];

  const jsDay=new Date(viewYear,viewMonth,day).getDay();

  const dayLabel=DAY_NAMES[jsDay]+', '+MONTH_NAMES[viewMonth]+' '+day+', '+viewYear;

  let html=`<div class="overlay" id="overlay" onclick="overlayClick(event)"><div class="modal">

    <div class="modal-header"><div class="modal-date-label">Posting Day</div><div class="modal-heading">${dayLabel}</div></div>

    <div class="modal-body">

    <div class="sec-label">1 — Content pillar</div>

    <div class="pillar-grid">`;

  PILLARS.forEach(pp=>{

    const isSel=draft.pillarId===pp.id;

    html+=`<button class="pb${isSel?' sel':''}" style="${isSel?`background:${pp.bg};color:${pp.fg};border-color:${pp.border}`:''}" onclick="selectPillar('${pp.id}',${day})">${pp.label}</button>`;

  });

  html+=`</div>`;

  if(draft.pillarId){

    html+=`<div class="sec-label">2 — Content copy</div>

    <textarea class="copy-area" id="copy-ta" placeholder="Write or paste your content copy here..." oninput="syncCopy(this.value)">${escHtml(draft.copy)}</textarea>

    <div class="char-count" id="char-count">${draft.copy.length} characters</div>

    <div class="sec-label">3 — Publishing time per channel</div>`;

      const chData=draft.channels[ch.id]||{enabled:true,time:ch.def};

      html+=`<div class="ch-row">

        <div class="ch-info">

          <div class="ch-icon" style="background:${ch.iconBg};color:${ch.iconColor}">${ch.icon}</div>

          <div><div class="ch-name">${ch.name}</div><div class="ch-platform">${ch.platform}</div></div>

      ${ex?`<button class="btn-remove" onclick="removeSlot()">Remove</button>`:''}

      <button class="btn-save" onclick="saveSlot()" ${!draft.pillarId?'disabled':''}>Save</button>

    </div>

  </div></div>`;

  document.getElementById('modal-area').innerHTML=html;

}

function overlayClick(e){if(e.target.id==='overlay')closeModal();}

function selectPillar(pid,day){draft.pillarId=pid;renderModal(day);}

function syncCopy(v){draft.copy=v;const cc=document.getElementById('char-count');if(cc)cc.textContent=v.length+' characters';}

function updateTime(chId,v){if(draft.channels[chId])draft.channels[chId].time=v;}

function toggleCh(chId,en,day){if(draft.channels[chId])draft.channels[chId].enabled=en;renderModal(day);}


async function saveSlot(){

  if(!draft.pillarId)return;

  const ta=document.getElementById('copy-ta');if(ta)draft.copy=ta.value;

  const entry={pillarId:draft.pillarId,copy:draft.copy,channels:JSON.parse(JSON.stringify(draft.channels))};

  assignments[modalDate]=entry;

  closeModal();renderCalendar();

  setStatus('Guardando...','loading');

  await postData({action:'save',dateKey:modalDate,...entry});

  setTimeout(async()=>{

    try{const d=await jsonpLoad();if(d.ok){assignments=d.entries||{};renderCalendar();}setStatus('Guardado · '+new Date().toLocaleTimeString(),'ok');}

    catch(e){setStatus('Guardado (sin confirmación)','ok');}

  },1800);

}


async function removeSlot(){

  const key=modalDate;

  delete assignments[key];

  closeModal();renderCalendar();

  await postData({action:'remove',dateKey:key});

  setStatus('Eliminado · '+new Date().toLocaleTimeString(),'ok');

}


function closeModal(){document.getElementById('modal-area').innerHTML='';modalDate=null;}

function changeMonth(dir){viewMonth+=dir;if(viewMonth>11){viewMonth=0;viewYear++;}if(viewMonth<0){viewMonth=11;viewYear--;}renderCalendar();}


async function resetAll(){

  if(confirm('¿Resetear TODO el contenido del equipo? No se puede deshacer.')){

    assignments={};renderCalendar();

    await postData({action:'reset'});

    setStatus('Reset exitoso','ok');

  }

}
